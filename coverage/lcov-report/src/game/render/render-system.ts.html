  const right = size + SELECTION_MARK_GAP;
  const top = -size - SELECTION_MARK_GAP;

  return new Graphics()
    .moveTo(left, top + arm)

    .moveTo(right - arm, top)
    .lineTo(right, top)
    .lineTo(right, top + arm)

}
